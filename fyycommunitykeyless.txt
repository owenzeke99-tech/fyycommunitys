local keyDuration = 86400 -- 86400 giây = 24 giờ

-- ================= Key Saving System ================= --
local function checkSavedKey()
    local success, result = pcall(function()
        if writefile and readfile and isfile and isfile(keyFileName) then
            local savedTime = tonumber(readfile(keyFileName))
            if savedTime and (os.time() - savedTime < keyDuration) then
                return true
            end
        end
        return false
    end)
    return success and result
end

local function saveKey()
    pcall(function()
        if writefile then
            writefile(keyFileName, tostring(os.time()))
        end
    end)
end

-- Tự động vượt qua bảng Key nếu đã lưu trong vòng 24h
if checkSavedKey() then
    local success, err = pcall(function()
        loadstring(game:HttpGet("https://raw.githubusercontent.com/napun87/stealanegg/refs/heads/main/fly.lua"))()
    end)
    if not success then
        warn("Lỗi khi tải script chính: " .. tostring(err))
    else
        return 
    end
end

-- ================= Smooth Hover Animations ================= --
local function addHoverEffect(button, normalColor, hoverColor)
    button.MouseEnter:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.2), {BackgroundColor3 = hoverColor}):Play()
    end)
    button.MouseLeave:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.2), {BackgroundColor3 = normalColor}):Play()
    end)
end

addHoverEffect(CheckBtn, Color3.fromRGB(105, 30, 180), Color3.fromRGB(130, 50, 220))

-- ================= Smooth Shake Animation ================= --
local function shakeUI()
    local originalPos = MainFrame.Position 
    local t = 0.05 
    local style = Enum.EasingStyle.Sine
    local dir = Enum.EasingDirection.InOut

    TweenService:Create(MainFrame, TweenInfo.new(t, style, dir), {Position = originalPos + UDim2.new(0, 15, 0, 0)}):Play()
    task.wait(t)
    TweenService:Create(MainFrame, TweenInfo.new(t, style, dir), {Position = originalPos - UDim2.new(0, 15, 0, 0)}):Play()
    task.wait(t)
    TweenService:Create(MainFrame, TweenInfo.new(t, style, dir), {Position = originalPos + UDim2.new(0, 8, 0, 0)}):Play()
    task.wait(t)
    TweenService:Create(MainFrame, TweenInfo.new(t, style, dir), {Position = originalPos - UDim2.new(0, 8, 0, 0)}):Play()
    task.wait(t)
    TweenService:Create(MainFrame, TweenInfo.new(t, style, dir), {Position = originalPos}):Play()
end

CheckBtn.MouseButton1Click:Connect(function()

    task.wait(0.1)
    TweenService:Create(CheckBtn, TweenInfo.new(0.1), {Size = UDim2.new(0, 140, 0, 35)}):Play()

    if KeyInput.Text == correctKey then
        StatusLabel.TextColor3 = Color3.fromRGB(85, 255, 127) 
        StatusLabel.Text = "Valid key! Loading script..."
        
        saveKey()
        
        task.wait(1.5)
        
        TweenService:Create(MainFrame, TweenInfo.new(0.5), {BackgroundTransparency = 1}):Play()
        TweenService:Create(MainStroke, TweenInfo.new(0.5), {Transparency = 1}):Play()
        
        for _, child in pairs(MainFrame:GetChildren()) do
            if child:IsA("GuiObject") then
                TweenService:Create(child, TweenInfo.new(0.5), {BackgroundTransparency = 1}):Play()
                if child:IsA("TextLabel") or child:IsA("TextBox") or child:IsA("TextButton") then
                    TweenService:Create(child, TweenInfo.new(0.5), {TextTransparency = 1}):Play()
                end
            end
        end
        task.wait(0.5)
        KeySystem:Destroy()
        
        -- Gọi script chính an toàn
        local success, err = pcall(function()
            loadstring(game:HttpGet("https://raw.githubusercontent.com/napun87/stealanegg/refs/heads/main/fly.lua"))()
        end)
        if not success then
            warn("Không thể chạy script chính: " .. tostring(err))
        end
    else
        StatusLabel.TextColor3 = Color3.fromRGB(255, 85, 85) 
        StatusLabel.Text = "Invalid Key! Please try again."
        
        KeyInput.Text = "" 
        shakeUI() 
        
        task.wait(2)
        StatusLabel.Text = ""
    end
end)
